<script>
  const prizes = [
    "🎁 خصم 10%",
    "🎉 خصم 20%",
    "💰 قسيمة شراء",
    "😢 لا يوجد جائزة",
    "🎁 هدية مجانية",
    "✨ جائزة مفاجأة",
    "🎯 نقطة مضاعفة",
    "💎 جائزة قيمة"
  ];

  const segmentAngle = 360 / prizes.length;
  const wheel = document.getElementById("wheel");

  // توليد UUID لكل مستخدم
  let userId = localStorage.getItem("userId");
  if (!userId) {
    userId = "user-" + Math.random().toString(36).substr(2, 16);
    localStorage.setItem("userId", userId);
  }

  const hasSpun = localStorage.getItem("hasSpun");
  if (hasSpun) {
    document.getElementById("spinBtn").disabled = true;
    document.getElementById("result").innerText =
      "❌ لقد قمتِ باللف مسبقًا\nجائزتك: " + hasSpun;
  }

  document.getElementById("spinBtn").onclick = function () {
    if (localStorage.getItem("hasSpun")) return;

    const prizeIndex = Math.floor(Math.random() * prizes.length);
    const rotation = 360 * 5 + prizeIndex * segmentAngle + segmentAngle / 2;
    wheel.style.transform = `rotate(${rotation}deg)`;

    setTimeout(() => {
      const prize = prizes[prizeIndex];
      document.getElementById("result").innerText =
        "🎉 مبروك! جائزتك: " + prize;

      // تسجيل النتيجة في Firebase مع معرف المستخدم
      db.collection("wheelResults").add({
        userId: userId,
        prize: prize,
        timestamp: new Date().toISOString()
      })
      .then(() => console.log("تم تسجيل النتيجة"))
      .catch((error) => console.error("خطأ في تسجيل النتيجة: ", error));

      localStorage.setItem("hasSpun", prize);
      document.getElementById("spinBtn").disabled = true;
    }, 4000);
  };
</script>
